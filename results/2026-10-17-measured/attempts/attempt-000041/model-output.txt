def merge_intervals(intervals):
    """
    Merge overlapping and touching intervals.
    
    Args:
        intervals: An iterable of pairs of integer endpoints
        
    Returns:
        A sorted list of disjoint, merged intervals as tuples
        
    Raises:
        ValueError: If a pair has wrong length, non-integer endpoints, or start > end
    """
    # Convert to list to support one-shot iterables and allow multiple passes
    interval_list = list(intervals)
    
    # Empty input returns empty list
    if not interval_list:
        return []
    
    # Validate and convert all intervals
    validated_intervals = []
    for interval in interval_list:
        # Check if it's a pair (length 2)
        try:
            interval_seq = tuple(interval)
        except TypeError:
            raise ValueError("Interval must be iterable")
        
        if len(interval_seq) != 2:
            raise ValueError(f"Interval must have exactly 2 endpoints, got {len(interval_seq)}")
        
        start, end = interval_seq
        
        # Check that endpoints are integers (bool is invalid, so check type strictly)
        if isinstance(start, bool) or isinstance(end, bool):
            raise ValueError("Endpoints must be integers, not bool")
        if not isinstance(start, int) or not isinstance(end, int):
            raise ValueError("Endpoints must be integers")
        
        # Check that start <= end
        if start > end:
            raise ValueError(f"Start {start} cannot be greater than end {end}")
        
        validated_intervals.append((start, end))
    
    # Sort intervals by start point, then by end point
    sorted_intervals = sorted(validated_intervals, key=lambda x: (x[0], x[1]))
    
    # Merge overlapping and touching intervals
    merged = [sorted_intervals[0]]
    
    for current_start, current_end in sorted_intervals[1:]:
        last_start, last_end = merged[-1]
        
        # Check if current interval overlaps or touches the last merged interval
        # Touching means current_start <= last_end (since intervals are closed)
        if current_start <= last_end:
            # Merge by extending the end point if necessary
            merged[-1] = (last_start, max(last_end, current_end))
        else:
            # No overlap, add as new interval
            merged.append((current_start, current_end))
    
    return merged
