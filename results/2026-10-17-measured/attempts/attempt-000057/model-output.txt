def tokenize(text: str) -> list[str]:
    tokens = []
    i = 0

    while i < len(text):
        char = text[i]

        if char.isspace():
            i += 1
        elif "0" <= char <= "9":
            start = i
            while i < len(text) and "0" <= text[i] <= "9":
                i += 1

            if i < len(text) and text[i] == ".":
                i += 1
                if i == len(text) or not ("0" <= text[i] <= "9"):
                    raise ValueError("Malformed number")
                while i < len(text) and "0" <= text[i] <= "9":
                    i += 1

            tokens.append(text[start:i])
        elif char in "+-*/()":
            tokens.append(char)
            i += 1
        else:
            raise ValueError(f"Invalid character: {char!r}")

    return tokens


def evaluate(text: str) -> float:
    tokens = tokenize(text)
    position = 0

    def expression() -> float:
        nonlocal position
        value = term()

        while position < len(tokens) and tokens[position] in ("+", "-"):
            operator = tokens[position]
            position += 1
            right = term()
            value = value + right if operator == "+" else value - right

        return value

    def term() -> float:
        nonlocal position
        value = unary()

        while position < len(tokens) and tokens[position] in ("*", "/"):
            operator = tokens[position]
            position += 1
            right = unary()
            value = value * right if operator == "*" else value / right

        return value

    def unary() -> float:
        nonlocal position
        sign = 1

        while position < len(tokens) and tokens[position] in ("+", "-"):
            if tokens[position] == "-":
                sign = -sign
            position += 1

        return sign * primary()

    def primary() -> float:
        nonlocal position

        if position >= len(tokens):
            raise ValueError("Expected a number or '('")

        token = tokens[position]
        if token == "(":
            position += 1
            value = expression()
            if position >= len(tokens) or tokens[position] != ")":
                raise ValueError("Unbalanced parentheses")
            position += 1
            return value

        if "0" <= token[0] <= "9":
            position += 1
            return float(token)

        raise ValueError("Expected a number or '('")

    result = expression()
    if position != len(tokens):
        raise ValueError("Unexpected trailing token")
    return result
