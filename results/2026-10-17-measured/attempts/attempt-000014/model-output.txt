import math
from collections import deque
from numbers import Real


class SlidingWindowLimiter:
    """Sliding-window rate limiter with independent history per hashable key."""

    def __init__(self, limit: int, window_s: float):
        if isinstance(limit, bool) or not isinstance(limit, int) or limit <= 0:
            raise ValueError("limit must be a positive integer")
        if isinstance(window_s, bool) or not isinstance(window_s, Real):
            raise ValueError("window_s must be a positive finite number")
        try:
            w = float(window_s)
        except (TypeError, ValueError, OverflowError):
            raise ValueError("window_s must be a positive finite number")
        if not math.isfinite(w) or w <= 0:
            raise ValueError("window_s must be a positive finite number")
        self._limit = limit
        self._window = window_s
        # key -> deque of accepted timestamps
        self._hist = {}
        # key -> last seen time (accepted or denied)
        self._last = {}

    def allow(self, key, now: float) -> bool:
        # Validate the key is hashable before touching any state.
        hash(key)

        if isinstance(now, bool) or not isinstance(now, Real):
            raise ValueError("now must be a finite number")
        try:
            fnow = float(now)
        except (TypeError, ValueError, OverflowError):
            raise ValueError("now must be a finite number")
        if not math.isfinite(fnow):
            raise ValueError("now must be a finite number")

        last = self._last.get(key)
        if last is not None and now < last:
            raise ValueError("time must be nondecreasing per key")

        # Validation passed; now mutate state.
        self._last[key] = now
        dq = self._hist.get(key)
        if dq is None:
            dq = deque()
            self._hist[key] = dq

        cutoff = now - self._window
        while dq and dq[0] <= cutoff:
            dq.popleft()

        if len(dq) < self._limit:
            dq.append(now)
            return True
        return False
