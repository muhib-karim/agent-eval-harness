```python
import re


_ROMAN_NUMERALS = [
    (1000, "M"), (900, "CM"), (500, "D"), (400, "CD"),
    (100, "C"), (90, "XC"), (50, "L"), (40, "XL"),
    (10, "X"), (9, "IX"), (5, "V"), (4, "IV"),
    (1, "I"),
]

_ROMAN_VALUES = {
    "I": 1, "V": 5, "X": 10, "L": 50,
    "C": 100, "D": 500, "M": 1000,
}

_CANONICAL_ROMAN_RE = re.compile(
    r"^M{0,3}(CM|CD|D?C{0,3})(XC|XL|L?X{0,3})(IX|IV|V?I{0,3})$"
)


def to_roman(value: int) -> str:
    """Convert an integer (1..3999) to its canonical uppercase Roman numeral."""
    if type(value) is not int:
        raise ValueError("value must be an int")
    if not (1 <= value <= 3999):
        raise ValueError("value must be in range 1..3999")

    result = []
    remaining = value
    for amount, symbol in _ROMAN_NUMERALS:
        if remaining <= 0:
            break
        count, remaining = divmod(remaining, amount)
        if count:
            result.append(symbol * count)
    return "".join(result)


def from_roman(text: str) -> int:
    """Convert a canonical uppercase Roman numeral string to an integer."""
    if type(text) is not str:
        raise ValueError("text must be a str")
    if text == "":
        raise ValueError("text must not be empty")
    if not _CANONICAL_ROMAN_RE.match(text):
        raise ValueError("text is not a valid canonical Roman numeral")

    total = 0
    i = 0
    n = len(text)
    while i < n:
        current = _ROMAN_VALUES[text[i]]
        if i + 1 < n:
            next_val = _ROMAN_VALUES[text[i + 1]]
            if current < next_val:
                total += next_val - current
                i += 2
                continue
        total += current
        i += 1

    if not (1 <= total <= 3999):
        raise ValueError("value must be in range 1..3999")

    return total
```
