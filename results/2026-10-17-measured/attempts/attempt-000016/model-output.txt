```python
"""Conversion between integers and canonical uppercase Roman numerals."""

_PAIRS = (
    (1000, "M"),
    (900, "CM"),
    (500, "D"),
    (400, "CD"),
    (100, "C"),
    (90, "XC"),
    (50, "L"),
    (40, "XL"),
    (10, "X"),
    (9, "IX"),
    (5, "V"),
    (4, "IV"),
    (1, "I"),
)

_SYMBOLS = {"I": 1, "V": 5, "X": 10, "L": 50, "C": 100, "D": 500, "M": 1000}

# Longest canonical numeral in range: MMMDCCCLXXXVIII (15 characters).
_MAX_LEN = 15


def to_roman(value: int) -> str:
    """Convert an integer in 1..3999 to a canonical uppercase Roman numeral."""
    if isinstance(value, bool) or not isinstance(value, int):
        raise ValueError("value must be an integer")
    if not 1 <= value <= 3999:
        raise ValueError("value must be in range 1..3999")
    remaining = int(value)
    parts = []
    for amount, symbol in _PAIRS:
        if remaining == 0:
            break
        count, remaining = divmod(remaining, amount)
        if count:
            parts.append(symbol * count)
    return "".join(parts)


def from_roman(text: str) -> int:
    """Convert a canonical uppercase Roman numeral to an integer in 1..3999."""
    if not isinstance(text, str):
        raise ValueError("text must be a string")
    if not text:
        raise ValueError("text must be nonempty")
    if len(text) > _MAX_LEN:
        raise ValueError("invalid Roman numeral")

    values = []
    for ch in text:
        v = _SYMBOLS.get(ch)
        if v is None:
            raise ValueError(f"invalid Roman numeral symbol: {ch!r}")
        values.append(v)

    total = 0
    for i, v in enumerate(values):
        if i + 1 < len(values) and v < values[i + 1]:
            total -= v
        else:
            total += v

    if not 1 <= total <= 3999:
        raise ValueError("invalid Roman numeral")

    # Canonical form check: rejects invalid repetitions and
    # noncanonical subtractive forms.
    if to_roman(total) != text:
        raise ValueError("noncanonical Roman numeral")
    return total
```
