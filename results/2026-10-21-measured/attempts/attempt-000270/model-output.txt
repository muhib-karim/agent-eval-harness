def levenshtein(a: str, b: str, max_distance: int | None = None) -> int:
    if not isinstance(a, str) or not isinstance(b, str):
        raise TypeError("a and b must be strings")
    if max_distance is not None and (
        isinstance(max_distance, bool)
        or not isinstance(max_distance, int)
        or max_distance < 0
    ):
        raise ValueError("max_distance must be a non-negative int or None")

    # Remove matching ends without making copies of either string.
    start = 0
    limit = min(len(a), len(b))
    while start < limit and a[start] == b[start]:
        start += 1

    a_end, b_end = len(a), len(b)
    while a_end > start and b_end > start and a[a_end - 1] == b[b_end - 1]:
        a_end -= 1
        b_end -= 1

    n, m = a_end - start, b_end - start
    if n < m:
        a, b = b, a
        n, m = m, n

    # The remaining part of b is the shorter one, so rows use O(m) space.
    if max_distance is not None and n - m > max_distance:
        return max_distance + 1
    if m == 0:
        return n

    if max_distance is None or max_distance >= n:
        previous = list(range(m + 1))
        current = [0] * (m + 1)

        for i in range(1, n + 1):
            current[0] = i
            char = a[start + i - 1]
            for j in range(1, m + 1):
                current[j] = min(
                    previous[j] + 1,
                    current[j - 1] + 1,
                    previous[j - 1] + (char != b[start + j - 1]),
                )
            previous, current = current, previous

        return previous[m]

    # Only cells within max_distance of the diagonal can contribute to
    # a result at or below the threshold.
    k = max_distance
    too_far = k + 1
    previous = [too_far] * (m + 1)
    current = [too_far] * (m + 1)
    previous_hi = min(m, k)
    for j in range(previous_hi + 1):
        previous[j] = j
    previous_lo = 0

    for i in range(1, n + 1):
        lo = max(0, i - k)
        hi = min(m, i + k)
        if lo > hi:
            return too_far

        row_min = too_far
        if lo == 0:
            current[0] = i
            row_min = min(row_min, i)

        char = a[start + i - 1]
        for j in range(max(1, lo), hi + 1):
            deletion = previous[j] + 1 if j <= previous_hi else too_far
            insertion = current[j - 1] + 1 if j > lo else too_far
            substitution = (
                previous[j - 1] + (char != b[start + j - 1])
                if previous_lo <= j - 1 <= previous_hi
                else too_far
            )
            value = min(deletion, insertion, substitution)
            current[j] = value
            if value < row_min:
                row_min = value

        if row_min > k:
            return too_far
        previous, current = current, previous
        previous_lo, previous_hi = lo, hi

    return min(previous[m], too_far)
