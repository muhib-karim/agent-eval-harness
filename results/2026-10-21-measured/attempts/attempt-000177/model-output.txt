"""
solution.py

A thread-unsafe (but simple and explicit) LRU cache with per-key TTL expiry.

LRUCache(capacity: int, ttl: float, clock=time.monotonic)
"""

import math
import time
from collections import OrderedDict


class LRUCache:
    """
    A Least-Recently-Used cache with a fixed capacity and a time-to-live (TTL)
    for each entry.

    - capacity: maximum number of *live* entries allowed before eviction.
    - ttl: number of time units (as measured by `clock`) an entry remains
      live after being inserted/updated via put().
    - clock: a zero-argument callable returning a nondecreasing timestamp.
      Defaults to time.monotonic.

    Semantics:
    - put(key, value) inserts/updates a key and sets its expiry to
      clock() + ttl. It also marks the key as most-recently-used.
    - get(key, default=None) returns the live value for key (or default if
      missing/expired), marking the key as most-recently-used. It does NOT
      extend the TTL.
    - An entry is considered expired once clock() >= its expiry timestamp.
    - Expired entries are purged before any eviction decision and before
      computing len(cache).
    - When the number of live entries exceeds capacity, the least recently
      used live entry is evicted.
    - len(cache) returns only the count of live (non-expired) entries.
    """

    def __init__(self, capacity, ttl, clock=time.monotonic):
        if isinstance(capacity, bool) or not isinstance(capacity, int) or capacity <= 0:
            raise ValueError("capacity must be a positive integer (not bool)")

        if isinstance(ttl, bool) or not isinstance(ttl, (int, float)):
            raise ValueError("ttl must be a positive finite number")
        if not math.isfinite(ttl) or ttl <= 0:
            raise ValueError("ttl must be a positive finite number")

        if not callable(clock):
            raise ValueError("clock must be callable")

        self.capacity = capacity
        self.ttl = float(ttl)
        self.clock = clock

        # Maps key -> (value, expiry_timestamp)
        # Order of the OrderedDict reflects recency: least-recently-used
        # entries are at the front, most-recently-used at the back.
        self._data = OrderedDict()

    def _purge_expired(self):
        """Remove all entries whose expiry has passed (expiry <= now)."""
        now = self.clock()
        expired_keys = [k for k, (_, expiry) in self._data.items() if now >= expiry]
        for k in expired_keys:
            del self._data[k]

    def put(self, key, value):
        """Insert or update `key` with `value`, resetting its TTL and
        marking it as most-recently-used."""
        self._purge_expired()

        expiry = self.clock() + self.ttl

        if key in self._data:
            # Remove first so re-insertion places it at the end (most recent).
            del self._data[key]

        self._data[key] = (value, expiry)
        self._data.move_to_end(key)

        # Purge again in case time has advanced and freed up space,
        # then evict if still over capacity.
        if len(self._data) > self.capacity:
            self._purge_expired()
            while len(self._data) > self.capacity:
                # Evict least recently used (front of the OrderedDict).
                self._data.popitem(last=False)

    def get(self, key, default=None):
        """Return the live value for `key`, or `default` if missing/expired.
        Marks the key as most-recently-used but does not extend its TTL."""
        self._purge_expired()

        if key not in self._data:
            return default

        value, expiry = self._data[key]
        self._data.move_to_end(key)
        return value

    def __len__(self):
        """Return the number of live (non-expired) entries."""
        self._purge_expired()
        return len(self._data)

    def __contains__(self, key):
        self._purge_expired()
        return key in self._data

    def __repr__(self):
        self._purge_expired()
        return f"LRUCache(capacity={self.capacity}, ttl={self.ttl}, size={len(self._data)})"
