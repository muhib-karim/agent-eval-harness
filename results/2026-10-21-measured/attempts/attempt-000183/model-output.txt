```python
import re

_ROMAN_NUMERALS = [
    (1000, 'M'),
    (900, 'CM'),
    (500, 'D'),
    (400, 'CD'),
    (100, 'C'),
    (90, 'XC'),
    (50, 'L'),
    (40, 'XL'),
    (10, 'X'),
    (9, 'IX'),
    (5, 'V'),
    (4, 'IV'),
    (1, 'I'),
]

_ROMAN_PATTERN = re.compile(
    r'^M{0,3}(CM|CD|D?C{0,3})(XC|XL|L?X{0,3})(IX|IV|V?I{0,3})$'
)

_ROMAN_VALUES = {
    'I': 1,
    'V': 5,
    'X': 10,
    'L': 50,
    'C': 100,
    'D': 500,
    'M': 1000,
}


def to_roman(value: int) -> str:
    """Convert an integer in range 1..3999 to a canonical uppercase Roman numeral."""
    if isinstance(value, bool) or not isinstance(value, int):
        raise ValueError("value must be an int (not bool) in range 1..3999")
    if value < 1 or value > 3999:
        raise ValueError("value must be in range 1..3999")

    result = []
    remaining = value
    for num, symbol in _ROMAN_NUMERALS:
        if remaining <= 0:
            break
        count, remaining = divmod(remaining, num)
        if count:
            result.append(symbol * count)
    return ''.join(result)


def from_roman(text: str) -> int:
    """Convert a canonical uppercase Roman numeral string to an integer."""
    if not isinstance(text, str):
        raise ValueError("text must be a str")
    if text == '':
        raise ValueError("text must be nonempty")

    # Reject whitespace, lowercase, or any character not in the valid set.
    for ch in text:
        if ch not in _ROMAN_VALUES:
            raise ValueError(f"invalid character in Roman numeral: {ch!r}")

    if not _ROMAN_PATTERN.match(text):
        raise ValueError(f"not a canonical Roman numeral: {text!r}")

    # Parse the value using standard left-to-right subtraction rule.
    total = 0
    i = 0
    n = len(text)
    while i < n:
        current = _ROMAN_VALUES[text[i]]
        if i + 1 < n:
            next_val = _ROMAN_VALUES[text[i + 1]]
            if current < next_val:
                total += next_val - current
                i += 2
                continue
        total += current
        i += 1

    if total < 1 or total > 3999:
        raise ValueError("value out of range 1..3999")

    return total
```
