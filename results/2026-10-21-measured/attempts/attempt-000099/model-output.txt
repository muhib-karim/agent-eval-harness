def tokenize(text: str) -> list[str]:
    tokens = []
    i = 0
    n = len(text)
    digits = "0123456789"
    whitespace = " \t\n\r\f\v"
    while i < n:
        c = text[i]
        if c in whitespace:
            i += 1
        elif c in "+-*/()":
            tokens.append(c)
            i += 1
        elif c in digits:
            j = i
            while j < n and text[j] in digits:
                j += 1
            if j < n and text[j] == ".":
                k = j + 1
                if k >= n or text[k] not in digits:
                    raise ValueError("malformed number")
                while k < n and text[k] in digits:
                    k += 1
                j = k
            tokens.append(text[i:j])
            i = j
        else:
            raise ValueError(f"invalid character {c!r}")
    return tokens


def evaluate(text: str) -> float:
    tokens = tokenize(text)
    if not tokens:
        raise ValueError("empty expression")
    pos = 0

    def peek():
        return tokens[pos] if pos < len(tokens) else None

    def parse_expr():
        nonlocal pos
        value = parse_term()
        while peek() in ("+", "-"):
            op = tokens[pos]
            pos += 1
            right = parse_term()
            value = value + right if op == "+" else value - right
        return value

    def parse_term():
        nonlocal pos
        value = parse_unary()
        while peek() in ("*", "/"):
            op = tokens[pos]
            pos += 1
            right = parse_unary()
            if op == "*":
                value = value * right
            else:
                value = value / right  # raises ZeroDivisionError
        return value

    def parse_unary():
        nonlocal pos
        sign = 1
        while peek() in ("+", "-"):
            if tokens[pos] == "-":
                sign = -sign
            pos += 1
        return sign * parse_primary()

    def parse_primary():
        nonlocal pos
        t = peek()
        if t is None:
            raise ValueError("unexpected end of expression")
        if t == "(":
            pos += 1
            value = parse_expr()
            if peek() != ")":
                raise ValueError("unbalanced parentheses")
            pos += 1
            return value
        if t[0] in "0123456789":
            pos += 1
            return float(t)
        raise ValueError(f"unexpected token {t!r}")

    result = parse_expr()
    if pos != len(tokens):
        raise ValueError("trailing tokens")
    return float(result)
