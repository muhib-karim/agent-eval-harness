def get_path(data, path: str, default=None):
    if not isinstance(path, str):
        raise TypeError("path must be a string")
    if path == "" or path == "$":
        return data

    n = len(path)
    ident_start = set("ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz_")
    digits = set("0123456789")
    ident_chars = ident_start | digits

    tokens = []  # ("k", name) or ("i", int)

    def read_ident(i):
        if i >= n or path[i] not in ident_start:
            raise ValueError("malformed path: expected identifier at position %d" % i)
        j = i + 1
        while j < n and path[j] in ident_chars:
            j += 1
        tokens.append(("k", path[i:j]))
        return j

    def read_bracket(i):
        # path[i] == "["
        j = i + 1
        k = j
        while k < n and path[k] in digits:
            k += 1
        if k == j or k >= n or path[k] != "]":
            raise ValueError("malformed index at position %d" % i)
        tokens.append(("i", int(path[j:k])))
        return k + 1

    i = 0
    if path[0] == "$":
        i = 1
        if path[i] == ".":
            i = read_ident(i + 1)
        elif path[i] != "[":
            raise ValueError("malformed path after '$'")
    elif path[0] != "[":
        i = read_ident(0)

    while i < n:
        c = path[i]
        if c == "[":
            i = read_bracket(i)
        elif c == ".":
            i = read_ident(i + 1)
        else:
            raise ValueError("malformed path at position %d" % i)

    cur = data
    for kind, val in tokens:
        if kind == "k":
            if isinstance(cur, dict) and val in cur:
                cur = cur[val]
            else:
                return default
        else:
            if isinstance(cur, list) and val < len(cur):
                cur = cur[val]
            else:
                return default
    return cur
