```python
import re

_SEMVER_RE = re.compile(
    r"""^
    (0|[1-9]\d*)\.          # major
    (0|[1-9]\d*)\.          # minor
    (0|[1-9]\d*)            # patch
    (?:-(
        (?:0|[1-9]\d*|\d*[a-zA-Z-][0-9a-zA-Z-]*)
        (?:\.(?:0|[1-9]\d*|\d*[a-zA-Z-][0-9a-zA-Z-]*))*
    ))?
    (?:\+(
        [0-9a-zA-Z-]+(?:\.[0-9a-zA-Z-]+)*
    ))?
    $""",
    re.VERBOSE,
)


def parse(text: str) -> tuple:
    if not isinstance(text, str):
        raise ValueError("version must be a string")

    m = _SEMVER_RE.match(text)
    if not m:
        raise ValueError(f"invalid semantic version: {text!r}")

    major_s, minor_s, patch_s, pre_s, build_s = m.groups()

    major = int(major_s)
    minor = int(minor_s)
    patch = int(patch_s)

    if pre_s:
        prerelease = tuple(
            int(ident) if ident.isdigit() else ident
            for ident in pre_s.split(".")
        )
    else:
        prerelease = ()

    if build_s:
        build = tuple(build_s.split("."))
    else:
        build = ()

    return (major, minor, patch, prerelease, build)


def _compare_identifiers(pre_a: tuple, pre_b: tuple) -> int:
    if not pre_a and not pre_b:
        return 0
    if not pre_a:
        # a has no prerelease -> a is greater
        return 1
    if not pre_b:
        # b has no prerelease -> b is greater
        return -1

    for ia, ib in zip(pre_a, pre_b):
        if ia == ib:
            continue
        a_is_num = isinstance(ia, int)
        b_is_num = isinstance(ib, int)
        if a_is_num and b_is_num:
            return -1 if ia < ib else 1
        if a_is_num and not b_is_num:
            return -1
        if not a_is_num and b_is_num:
            return 1
        # both strings
        return -1 if ia < ib else 1

    if len(pre_a) < len(pre_b):
        return -1
    if len(pre_a) > len(pre_b):
        return 1
    return 0


def compare(a: str, b: str) -> int:
    pa = parse(a)
    pb = parse(b)

    core_a = pa[:3]
    core_b = pb[:3]

    if core_a != core_b:
        return -1 if core_a < core_b else 1

    return _compare_identifiers(pa[3], pb[3])
```
