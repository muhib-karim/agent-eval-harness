```python
"""Streaming mean and sample variance (Welford), mergeable across shards."""

import math


class RunningStats:
    def __init__(self):
        self._n = 0
        self._mean = 0.0
        self._m2 = 0.0

    def push(self, x):
        if isinstance(x, bool) or not isinstance(x, (int, float)) or not math.isfinite(x):
            raise ValueError("x must be a finite number")
        self._n += 1
        delta = x - self._mean
        self._mean += delta / self._n
        self._m2 += delta * (x - self._mean)

    @property
    def count(self):
        return self._n

    @property
    def mean(self):
        if self._n < 1:
            raise ValueError("mean needs at least one value")
        return self._mean

    @property
    def variance(self):
        if self._n < 2:
            raise ValueError("variance needs at least two values")
        return self._m2 / (self._n - 1)

    @property
    def stdev(self):
        return math.sqrt(self.variance)

    def merge(self, other):
        result = RunningStats()
        n = self._n + other._n
        if n == 0:
            return result
        delta = other._mean - self._mean
        result._n = n
        result._mean = self._mean + delta * other._n / n
        result._m2 = self._m2 + other._m2 + delta * delta * self._n * other._n / n
        return result
```

The bug was in the `merge()` method. When merging two RunningStats objects with different means, the formula for combining the sum of squared deviations (`_m2`) was incomplete.

The correct formula for merging two datasets is:
```
M2_combined = M2_a + M2_b + (mean_b - mean_a)² * n_a * n_b / (n_a + n_b)
```

The original code was missing the last term: `delta * delta * self._n * other._n / n`

This term accounts for the additional variance introduced by the difference in means between the two datasets. Without it, the merged variance is significantly underestimated when the two inputs have different means.
