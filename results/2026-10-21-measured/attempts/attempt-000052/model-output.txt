The root cause: `_right` returns the index **one past** the last element `<= target` (the upper bound), not the index of the last match itself. `find_last` then checked that past-the-end position, which is either out of range or holds a larger value, so it always returned `-1`. The fix is to step back one position and check that index instead (guarding against `index < 0` for an empty list or a target smaller than everything).

```python
"""Binary-search helpers over an ascending list."""


def _left(items, target):
    lo, hi = 0, len(items)
    while lo < hi:
        mid = (lo + hi) // 2
        if items[mid] < target:
            lo = mid + 1
        else:
            hi = mid
    return lo


def _right(items, target):
    lo, hi = 0, len(items)
    while lo < hi:
        mid = (lo + hi) // 2
        if items[mid] <= target:
            lo = mid + 1
        else:
            hi = mid
    return lo


def find_first(items, target):
    """Index of the first element equal to target, or -1."""
    index = _left(items, target)
    if index < len(items) and items[index] == target:
        return index
    return -1


def find_last(items, target):
    """Index of the last element equal to target, or -1."""
    # _right gives the position just past the last element <= target,
    # so the candidate for the last match is one position before it.
    index = _right(items, target) - 1
    if index >= 0 and items[index] == target:
        return index
    return -1


def count_range(items, low, high):
    """Number of elements x with low <= x <= high."""
    if low > high:
        return 0
    return _right(items, high) - _left(items, low)
```
