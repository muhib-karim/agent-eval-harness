```python
import math
import time
from collections import OrderedDict


class LRUCache:
    """An LRU cache whose entries expire ``ttl`` time units after being put.

    Entries are stored in an OrderedDict ordered from least recently used
    (front) to most recently used (back).  Each value is stored together with
    its absolute expiry time as measured by the injected ``clock``.
    """

    __slots__ = ("_capacity", "_ttl", "_clock", "_data")

    def __init__(self, capacity, ttl, clock=time.monotonic):
        # capacity: positive integer, explicitly rejecting bool
        if isinstance(capacity, bool) or not isinstance(capacity, int):
            raise ValueError("capacity must be a positive integer")
        if capacity <= 0:
            raise ValueError("capacity must be a positive integer")

        # ttl: positive, finite real number (reject bool as well)
        if isinstance(ttl, bool) or not isinstance(ttl, (int, float)):
            raise ValueError("ttl must be a positive finite number")
        try:
            ttl_f = float(ttl)
        except (TypeError, ValueError, OverflowError):
            raise ValueError("ttl must be a positive finite number")
        if not math.isfinite(ttl_f) or ttl_f <= 0:
            raise ValueError("ttl must be a positive finite number")

        if not callable(clock):
            raise ValueError("clock must be callable")

        self._capacity = capacity
        self._ttl = ttl_f
        self._clock = clock
        self._data = OrderedDict()  # key -> (value, expiry)

    # ------------------------------------------------------------------ #
    # Internal helpers
    # ------------------------------------------------------------------ #
    def _purge(self, now=None):
        """Remove every entry whose expiry has been reached."""
        if now is None:
            now = self._clock()
        expired = [k for k, (_, exp) in self._data.items() if now >= exp]
        for k in expired:
            del self._data[k]

    # ------------------------------------------------------------------ #
    # Public API
    # ------------------------------------------------------------------ #
    @property
    def capacity(self):
        return self._capacity

    @property
    def ttl(self):
        return self._ttl

    def put(self, key, value):
        """Insert or update ``key`` with ``value`` and reset its expiry."""
        now = self._clock()
        expiry = now + self._ttl
        if key in self._data:
            self._data[key] = (value, expiry)
            self._data.move_to_end(key)
        else:
            self._data[key] = (value, expiry)
        # Purge expired entries before evicting live ones.
        if len(self._data) > self._capacity:
            self._purge(now)
            while len(self._data) > self._capacity:
                self._data.popitem(last=False)

    def get(self, key, default=None):
        """Return the live value for ``key`` (marking it most recently used)."""
        entry = self._data.get(key)
        if entry is None and key not in self._data:
            return default
        value, expiry = entry
        if self._clock() >= expiry:
            del self._data[key]
            return default
        self._data.move_to_end(key)
        return value

    def delete(self, key):
        """Remove ``key`` if present; return True if something was removed."""
        if key in self._data:
            del self._data[key]
            return True
        return False

    def clear(self):
        self._data.clear()

    def __contains__(self, key):
        entry = self._data.get(key)
        if entry is None and key not in self._data:
            return False
        _, expiry = entry
        if self._clock() >= expiry:
            del self._data[key]
            return False
        return True

    def __len__(self):
        self._purge()
        return len(self._data)

    def keys(self):
        """Live keys in order from least to most recently used."""
        self._purge()
        return list(self._data.keys())

    def __repr__(self):
        return "LRUCache(capacity=%r, ttl=%r, size=%d)" % (
            self._capacity,
            self._ttl,
            len(self),
        )
```
