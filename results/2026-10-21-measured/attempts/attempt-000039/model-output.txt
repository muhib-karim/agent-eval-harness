```python
"""Roman numeral conversion utilities."""

_PAIRS = (
    (1000, "M"),
    (900, "CM"),
    (500, "D"),
    (400, "CD"),
    (100, "C"),
    (90, "XC"),
    (50, "L"),
    (40, "XL"),
    (10, "X"),
    (9, "IX"),
    (5, "V"),
    (4, "IV"),
    (1, "I"),
)

_SYMBOLS = {"I": 1, "V": 5, "X": 10, "L": 50, "C": 100, "D": 500, "M": 1000}


def to_roman(value: int) -> str:
    """Convert an integer in 1..3999 to a canonical uppercase Roman numeral."""
    if isinstance(value, bool) or not isinstance(value, int):
        raise ValueError("value must be an integer")
    if not 1 <= value <= 3999:
        raise ValueError("value must be in range 1..3999")
    n = int(value)
    parts = []
    for amount, symbol in _PAIRS:
        count, n = divmod(n, amount)
        if count:
            parts.append(symbol * count)
    return "".join(parts)


def from_roman(text: str) -> int:
    """Convert a canonical uppercase Roman numeral to an integer in 1..3999."""
    if not isinstance(text, str):
        raise ValueError("text must be a string")
    s = str(text)
    if not s:
        raise ValueError("empty Roman numeral")
    for ch in s:
        if ch not in _SYMBOLS:
            raise ValueError("invalid Roman numeral character: %r" % ch)

    total = 0
    length = len(s)
    for i, ch in enumerate(s):
        v = _SYMBOLS[ch]
        if i + 1 < length and _SYMBOLS[s[i + 1]] > v:
            total -= v
        else:
            total += v

    if not 1 <= total <= 3999:
        raise ValueError("Roman numeral out of range")
    if to_roman(total) != s:
        raise ValueError("noncanonical Roman numeral: %r" % s)
    return total
```
