```python
import math
import time
from collections import OrderedDict
from numbers import Real


class LRUCache:
    def __init__(self, capacity: int, ttl: float, clock=time.monotonic):
        if isinstance(capacity, bool) or not isinstance(capacity, int) or capacity <= 0:
            raise ValueError("capacity must be a positive integer")
        try:
            if not isinstance(ttl, Real) or not math.isfinite(ttl) or not ttl > 0:
                raise ValueError("ttl must be positive and finite")
        except (TypeError, OverflowError):
            raise ValueError("ttl must be positive and finite")
        self._capacity = capacity
        self._ttl = ttl
        self._clock = clock
        # key -> value, ordered by recency (oldest first)
        self._data = OrderedDict()
        # key -> expiry, ordered by expiry (earliest first). Because the clock is
        # nondecreasing and ttl is constant, re-putting a key moves it to the end.
        self._expiry = OrderedDict()

    def _purge(self, now):
        expiry = self._expiry
        while expiry:
            key = next(iter(expiry))
            if expiry[key] <= now:
                del expiry[key]
                del self._data[key]
            else:
                break

    def put(self, key, value):
        now = self._clock()
        self._purge(now)
        if key in self._data:
            self._data[key] = value
            self._data.move_to_end(key)
        else:
            self._data[key] = value
        self._expiry[key] = now + self._ttl
        self._expiry.move_to_end(key)
        while len(self._data) > self._capacity:
            old_key, _ = self._data.popitem(last=False)
            del self._expiry[old_key]

    def get(self, key, default=None):
        now = self._clock()
        self._purge(now)
        if key in self._data:
            self._data.move_to_end(key)
            return self._data[key]
        return default

    def __contains__(self, key):
        self._purge(self._clock())
        return key in self._data

    def __len__(self):
        self._purge(self._clock())
        return len(self._data)
```
