def topo_sort(graph):
    """
    Topologically sort a directed acyclic graph.
    
    Args:
        graph: A dict mapping each node to an iterable of nodes it depends on.
               Dependencies are nodes that must come before the dependent node.
    
    Returns:
        A list of all nodes in topological order, with dependencies before dependents.

    Raises:

    """
    # Collect all nodes from both keys and dependency values
    all_nodes = set()
    for node, dependencies in graph.items():

            all_nodes.add(dep)
    
    # Build reverse graph and calculate in-degrees
    reverse_graph = {node: [] for node in all_nodes}
    in_degree = {node: 0 for node in all_nodes}

        deps_set = set(dependencies)
        in_degree[node] = len(deps_set)

            reverse_graph[dep].append(node)
    
    # Find all nodes with no dependencies and sort them
