```python
import threading


class InsufficientFunds(ValueError):
    pass


def _valid_int(x):
    return isinstance(x, int) and not isinstance(x, bool)


class Ledger:
    def __init__(self):
        # A single lock guards all state. Every operation takes just this one
        # lock, so transfers in opposite directions cannot deadlock and total()
        # always sees a consistent snapshot.
        self._lock = threading.Lock()
        self._balances = {}
        self._history = {}

    @staticmethod
    def _check_amount(amount):
        if not _valid_int(amount) or amount <= 0:
            raise ValueError("amount must be a positive int")

    def _get(self, account):
        if not isinstance(account, str) or account not in self._balances:
            raise KeyError(account)

    def open(self, account, balance=0):
        if not isinstance(account, str) or not account:
            raise ValueError("account id must be a non-empty str")
        if not _valid_int(balance) or balance < 0:
            raise ValueError("balance must be a non-negative int")
        with self._lock:
            if account in self._balances:
                raise ValueError("account already exists")
            self._balances[account] = balance
            self._history[account] = [("open", balance)]

    def balance(self, account):
        with self._lock:
            self._get(account)
            return self._balances[account]

    def deposit(self, account, amount):
        self._check_amount(amount)
        with self._lock:
            self._get(account)
            self._balances[account] += amount
            self._history[account].append(("deposit", amount))

    def withdraw(self, account, amount):
        self._check_amount(amount)
        with self._lock:
            self._get(account)
            if self._balances[account] < amount:
                raise InsufficientFunds("insufficient funds")
            self._balances[account] -= amount
            self._history[account].append(("withdraw", amount))

    def transfer(self, src, dst, amount):
        self._check_amount(amount)
        if isinstance(src, str) and isinstance(dst, str) and src == dst:
            raise ValueError("cannot transfer to the same account")
        with self._lock:
            self._get(src)
            self._get(dst)
            if self._balances[src] < amount:
                raise InsufficientFunds("insufficient funds")
            self._balances[src] -= amount
            self._balances[dst] += amount
            self._history[src].append(("transfer_out", amount))
            self._history[dst].append(("transfer_in", amount))

    def total(self):
        with self._lock:
            return sum(self._balances.values())

    def history(self, account):
        with self._lock:
            self._get(account)
            return list(self._history[account])
```
