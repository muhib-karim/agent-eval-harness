"""
solution.py

A simple LRU (Least Recently Used) cache with TTL (time-to-live) expiration.

Only standard library is used.
"""

import math
import time
from collections import OrderedDict


class LRUCache:
    """
    A cache that evicts the Least Recently Used entry once capacity is
    exceeded, and treats entries as expired once `clock() >= expiry`.

    - capacity: positive int (not bool).
    - ttl: positive, finite number (int or float).
    - clock: a zero-argument callable returning a nondecreasing float,
      defaults to time.monotonic.
    """

    def __init__(self, capacity, ttl, clock=time.monotonic):
        if isinstance(capacity, bool) or not isinstance(capacity, int) or capacity <= 0:
            raise ValueError("capacity must be a positive integer (not bool)")

        if isinstance(ttl, bool) or not isinstance(ttl, (int, float)):
            raise ValueError("ttl must be a positive, finite number")
        if not math.isfinite(ttl) or ttl <= 0:
            raise ValueError("ttl must be positive and finite")

        self.capacity = capacity
        self.ttl = float(ttl)
        self.clock = clock

        # key -> (value, expiry_time)
        # OrderedDict maintains recency order: least-recently-used first.
        self._data = OrderedDict()

    def _purge_expired(self):
        """Remove all entries whose expiry time has passed."""
        now = self.clock()
        expired_keys = [k for k, (_, expiry) in self._data.items() if now >= expiry]
        for k in expired_keys:
            del self._data[k]

    def _evict_if_needed(self):
        """Evict least-recently-used entries until within capacity."""
        while len(self._data) > self.capacity:
            self._data.popitem(last=False)

    def put(self, key, value):
        """
        Insert or update `key` with `value`. Resets the entry's expiry to
        clock() + ttl and marks it as most recently used.
        """
        self._purge_expired()

        expiry = self.clock() + self.ttl

        # Remove existing entry (if any) so re-inserting places it at the
        # most-recently-used end.
        if key in self._data:
            del self._data[key]

        self._data[key] = (value, expiry)
        self._data.move_to_end(key)

        self._evict_if_needed()

    def get(self, key, default=None):
        """
        Return the live value for `key`, marking it as most recently used.
        Does NOT extend the TTL. Returns `default` if missing or expired.
        """
        self._purge_expired()

        if key not in self._data:
            return default

        value, expiry = self._data[key]
        self._data.move_to_end(key)
        return value

    def __len__(self):
        """Count only live (non-expired) entries."""
        self._purge_expired()
        return len(self._data)

    def __contains__(self, key):
        """Convenience: check for a live key without altering recency."""
        self._purge_expired()
        return key in self._data

    def __repr__(self):
        self._purge_expired()
        items = ", ".join(f"{k!r}: {v!r}" for k, (v, _) in self._data.items())
        return f"LRUCache(capacity={self.capacity}, ttl={self.ttl}, items={{{items}}})"
