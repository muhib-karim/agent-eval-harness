"""Merge closed integer intervals."""

from typing import Iterable, List, Tuple


def _is_int(value) -> bool:
    # bool is a subclass of int, so it has to be excluded explicitly.
    return isinstance(value, int) and not isinstance(value, bool)


def merge_intervals(intervals: Iterable) -> List[Tuple[int, int]]:
    """Return sorted, disjoint closed intervals.

    Intervals that overlap or share an endpoint are merged.
    Raises ValueError for malformed pairs, non-integer endpoints,
    or intervals with start > end. The input is never mutated.
    """
    if intervals is None:
        raise ValueError("intervals must be an iterable of pairs")
    try:
        iterator = iter(intervals)
    except TypeError:
        raise ValueError("intervals must be an iterable of pairs") from None

    pairs: List[Tuple[int, int]] = []
    for item in iterator:
        # Strings and bytes are iterable but are not valid pairs.
        if isinstance(item, (str, bytes, bytearray)):
            raise ValueError(f"invalid interval: {item!r}")
        try:
            # Copy into a new tuple so the original member is never modified.
            pair = tuple(item)
        except TypeError:
            raise ValueError(f"interval is not a pair: {item!r}") from None
        if len(pair) != 2:
            raise ValueError(f"interval must have exactly 2 endpoints: {item!r}")
        start, end = pair
        if not _is_int(start) or not _is_int(end):
            raise ValueError(f"interval endpoints must be integers: {item!r}")
        if start > end:
            raise ValueError(f"interval start > end: {item!r}")
        pairs.append((int(start), int(end)))

    if not pairs:
        return []

    pairs.sort()
    merged: List[Tuple[int, int]] = []
    cur_start, cur_end = pairs[0]
    for start, end in pairs[1:]:
        if start <= cur_end:  # overlapping or touching endpoints
            if end > cur_end:
                cur_end = end
        else:
            merged.append((cur_start, cur_end))
            cur_start, cur_end = start, end
    merged.append((cur_start, cur_end))
    return merged
