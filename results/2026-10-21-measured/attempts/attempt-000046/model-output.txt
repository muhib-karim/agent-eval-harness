`TextBuffer` keeps undo and redo stacks of edit groups. Each group is a list of `(kind, index, string)` records, and every single edit is stored as a one-item group.

The spec left three points open, so I chose these behaviours:

- **Calling `undo()`/`redo()` inside a `group()` block:** this raises `RuntimeError`.
- **Redo history inside a group:** the first real edit in a group clears the redo history right away.
- **Rollback when a block raises:** the text and the redo history go back to how they were when that block started. If a nested block raises and the outer block catches the exception, only the nested block's edits are rolled back.

```python
from contextlib import contextmanager


class TextBuffer:
    def __init__(self, text: str = ""):
        if not isinstance(text, str):
            raise ValueError("text must be a str")
        self._text = text
        self._undo = []   # list of groups; each group is a list of ops
        self._redo = []
        self._depth = 0
        self._current = None  # ops of the active outermost group

    @property
    def text(self) -> str:
        return self._text

    @staticmethod
    def _is_int(x):
        return isinstance(x, int) and not isinstance(x, bool)

    # --- low-level application ---
    def _apply(self, op):
        kind, idx, s = op
        if kind == "ins":
            self._text = self._text[:idx] + s + self._text[idx:]
        else:
            self._text = self._text[:idx] + self._text[idx + len(s):]

    def _revert(self, op):
        kind, idx, s = op
        if kind == "ins":
            self._text = self._text[:idx] + self._text[idx + len(s):]
        else:
            self._text = self._text[:idx] + s + self._text[idx:]

    def _record(self, op):
        self._apply(op)
        self._redo = []
        if self._depth > 0:
            self._current.append(op)
        else:
            self._undo.append([op])

    # --- public edit API ---
    def insert(self, index: int, s: str) -> None:
        if not self._is_int(index):
            raise ValueError("index must be an int")
        if not isinstance(s, str):
            raise ValueError("s must be a str")
        if index < 0 or index > len(self._text):
            raise ValueError("index out of range")
        if s == "":
            return
        self._record(("ins", index, s))

    def delete(self, start: int, end: int) -> None:
        if not self._is_int(start) or not self._is_int(end):
            raise ValueError("positions must be ints")
        if start < 0 or end > len(self._text) or start > end:
            raise ValueError("invalid range")
        if start == end:
            return
        self._record(("del", start, self._text[start:end]))

    def undo(self) -> bool:
        if self._depth > 0:
            raise RuntimeError("cannot undo inside a group")
        if not self._undo:
            return False
        grp = self._undo.pop()
        for op in reversed(grp):
            self._revert(op)
        self._redo.append(grp)
        return True

    def redo(self) -> bool:
        if self._depth > 0:
            raise RuntimeError("cannot redo inside a group")
        if not self._redo:
            return False
        grp = self._redo.pop()
        for op in grp:
            self._apply(op)
        self._undo.append(grp)
        return True

    @contextmanager
    def group(self):
        outermost = self._depth == 0
        if outermost:
            self._current = []
        start_count = len(self._current)
        saved_text = self._text
        saved_redo = list(self._redo)
        self._depth += 1
        try:
            yield self
        except BaseException:
            ops = self._current
            while len(ops) > start_count:
                self._revert(ops.pop())
            self._text = saved_text
            self._redo = saved_redo
            self._depth -= 1
            if outermost:
                self._current = None
            raise
        else:
            self._depth -= 1
            if outermost:
                ops = self._current
                self._current = None
                if ops:
                    self._undo.append(ops)
```
