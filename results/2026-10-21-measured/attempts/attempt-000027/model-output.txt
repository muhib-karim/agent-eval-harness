The ledger uses a single lock that every method takes, so every operation, including `total()`, sees a consistent state. Because there is only one lock, two transfers in opposite directions can never deadlock.

```python
import threading


class InsufficientFunds(ValueError):
    """Raised when an operation would make a balance negative."""
    pass


def _is_int(value):
    return isinstance(value, int) and not isinstance(value, bool)


class Ledger:
    def __init__(self):
        self._lock = threading.Lock()
        self._balances = {}
        self._history = {}

    # --- validation helpers -------------------------------------------------

    @staticmethod
    def _check_amount(amount):
        if not _is_int(amount) or amount <= 0:
            raise ValueError("amount must be a positive int")

    def _require(self, account):
        # Must be called with the lock held.
        try:
            return self._balances[account]
        except (KeyError, TypeError):
            raise KeyError(account)

    # --- public API ---------------------------------------------------------

    def open(self, account, balance=0):
        if not isinstance(account, str) or account == "":
            raise ValueError("account id must be a non-empty str")
        if not _is_int(balance) or balance < 0:
            raise ValueError("balance must be a non-negative int")
        with self._lock:
            if account in self._balances:
                raise ValueError("account already exists: %r" % (account,))
            self._balances[account] = balance
            self._history[account] = [("open", balance)]

    def balance(self, account):
        with self._lock:
            return self._require(account)

    def deposit(self, account, amount):
        self._check_amount(amount)
        with self._lock:
            current = self._require(account)
            self._balances[account] = current + amount
            self._history[account].append(("deposit", amount))

    def withdraw(self, account, amount):
        self._check_amount(amount)
        with self._lock:
            current = self._require(account)
            if current < amount:
                raise InsufficientFunds(
                    "insufficient funds in %r: balance %d, requested %d"
                    % (account, current, amount)
                )
            self._balances[account] = current - amount
            self._history[account].append(("withdraw", amount))

    def transfer(self, src, dst, amount):
        if src == dst:
            raise ValueError("source and destination must differ")
        self._check_amount(amount)
        with self._lock:
            src_bal = self._require(src)
            dst_bal = self._require(dst)
            if src_bal < amount:
                raise InsufficientFunds(
                    "insufficient funds in %r: balance %d, requested %d"
                    % (src, src_bal, amount)
                )
            self._balances[src] = src_bal - amount
            self._balances[dst] = dst_bal + amount
            self._history[src].append(("transfer_out", amount))
            self._history[dst].append(("transfer_in", amount))

    def total(self):
        with self._lock:
            return sum(self._balances.values())

    def history(self, account):
        with self._lock:
            self._require(account)
            return list(self._history[account])
```
