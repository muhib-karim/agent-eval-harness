_ROMAN_VALUES = (
    (1000, "M"),
    (900, "CM"),
    (500, "D"),
    (400, "CD"),
    (100, "C"),
    (90, "XC"),
    (50, "L"),
    (40, "XL"),
    (10, "X"),
    (9, "IX"),
    (5, "V"),
    (4, "IV"),
    (1, "I"),
)

_SYMBOL_VALUES = {
    "I": 1, "V": 5, "X": 10, "L": 50,
    "C": 100, "D": 500, "M": 1000,
}


def to_roman(value: int) -> str:
    if type(value) is not int or not 1 <= value <= 3999:
        raise ValueError("value must be an integer from 1 to 3999")

    parts = []
    for amount, symbol in _ROMAN_VALUES:
        count, value = divmod(value, amount)
        parts.append(symbol * count)
    return "".join(parts)


def from_roman(text: str) -> int:
    if not isinstance(text, str) or not text:
        raise ValueError("text must be a nonempty canonical Roman numeral")

    try:
        values = [_SYMBOL_VALUES[char] for char in text]
    except KeyError as exc:
        raise ValueError("invalid Roman numeral") from exc

    total = 0
    for index, value in enumerate(values):
        if index + 1 < len(values) and value < values[index + 1]:
            total -= value
        else:
            total += value

    if not 1 <= total <= 3999 or to_roman(total) != text:
        raise ValueError("invalid Roman numeral")
    return total
