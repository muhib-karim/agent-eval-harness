import re
from typing import Union

def tokenize(text: str) -> list[str]:
    """
    Tokenize a mathematical expression into a list of tokens.
    
    Valid tokens:
    - ASCII unsigned integers or decimals (digits on both sides of decimal point)
    - Single-character operators: + - * / ( )
    - Whitespace is ignored
    
    Raises ValueError for any other character or malformed numbers.
    Returns empty list for empty input.
    """
    if not text:
        return []
    
    tokens = []
    i = 0
    
    while i < len(text):
        # Skip whitespace
        if text[i].isspace():
            i += 1
            continue
        
        # Check for number (integer or decimal)
        if text[i].isdigit():
            start = i
            has_decimal = False
            
            # Collect digits and at most one decimal point
            while i < len(text) and (text[i].isdigit() or text[i] == '.'):
                if text[i] == '.':
                    if has_decimal:
                        # Multiple decimal points
                        raise ValueError(f"Malformed number: multiple decimal points")
                    has_decimal = True
                i += 1
            
            number_str = text[start:i]
            
            # Check if decimal number has digits on both sides
            if '.' in number_str:
                parts = number_str.split('.')
                if not parts[0] or not parts[1]:
                    raise ValueError(f"Malformed number: digits required on both sides of decimal point in {number_str}")
            
            tokens.append(number_str)
            continue
        
        # Check for single-character operators and parentheses
        if text[i] in '+-*/()':
            tokens.append(text[i])
            i += 1
            continue
        
        # Invalid character
        raise ValueError(f"Invalid character: {text[i]}")
    
    return tokens


def evaluate(text: str) -> float:
    """
    Evaluate a mathematical expression.
    
    Supports:
    - Parentheses for grouping
    - Unary + and - (including repeated)
    - Binary operators: + - * / with standard precedence
    - Left-associative operators
    
    Raises ValueError for:
    - Empty expressions
    - Adjacent operands
    - Trailing tokens
    - Unbalanced parentheses
    
    Raises ZeroDivisionError for division by zero.
    """
    tokens = tokenize(text)
    
    if not tokens:
        raise ValueError("Empty expression")
    
    pos = [0]  # Use list to allow modification in nested functions
    
    def parse_expression():
        """Parse addition and subtraction (lowest precedence)"""
        left = parse_term()
        
        while pos[0] < len(tokens) and tokens[pos[0]] in ['+', '-']:
            op = tokens[pos[0]]
            pos[0] += 1
            right = parse_term()
            if op == '+':
                left = left + right
            else:
                left = left - right
        
        return left
    
    def parse_term():
        """Parse multiplication and division (higher precedence)"""
        left = parse_unary()
        
        while pos[0] < len(tokens) and tokens[pos[0]] in ['*', '/']:
            op = tokens[pos[0]]
            pos[0] += 1
            right = parse_unary()
            if op == '*':
                left = left * right
            else:
                if right == 0:
                    raise ZeroDivisionError("Division by zero")
                left = left / right
        
        return left
    
    def parse_unary():
        """Parse unary operators and primary expressions"""
        # Handle unary + and -
        sign = 1
        while pos[0] < len(tokens) and tokens[pos[0]] in ['+', '-']:
            if tokens[pos[0]] == '-':
                sign *= -1
            pos[0] += 1
        
        value = parse_primary()
        return sign * value
    
    def parse_primary():
        """Parse numbers and parenthesized expressions"""
        if pos[0] >= len(tokens):
            raise ValueError("Unexpected end of expression")
        
        token = tokens[pos[0]]
        
        # Check for number
        if token[0].isdigit():
            pos[0] += 1
            return float(token)
        
        # Check for opening parenthesis
        if token == '(':
            pos[0] += 1
            value = parse_expression()
            
            if pos[0] >= len(tokens) or tokens[pos[0]] != ')':
                raise ValueError("Unbalanced parentheses: missing closing ')'")
            
            pos[0] += 1
            return value
        
        # Invalid token at this position
        raise ValueError(f"Unexpected token: {token}")
    
    result = parse_expression()
    
    # Check if all tokens have been consumed
    if pos[0] < len(tokens):
        raise ValueError(f"Trailing tokens: {tokens[pos[0]:]}")
    
    return result
