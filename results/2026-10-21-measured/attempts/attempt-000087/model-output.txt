```python
"""Roman numeral conversion utilities.

Provides:
    to_roman(value)  -> canonical uppercase Roman numeral for 1..3999
    from_roman(text) -> integer for a canonical uppercase Roman numeral
"""

import re

__all__ = ["to_roman", "from_roman"]

_MIN = 1
_MAX = 3999

# Ordered from largest to smallest; includes the subtractive pairs.
_VALUES = (
    (1000, "M"),
    (900, "CM"),
    (500, "D"),
    (400, "CD"),
    (100, "C"),
    (90, "XC"),
    (50, "L"),
    (40, "XL"),
    (10, "X"),
    (9, "IX"),
    (5, "V"),
    (4, "IV"),
    (1, "I"),
)

_SYMBOLS = {
    "I": 1,
    "V": 5,
    "X": 10,
    "L": 50,
    "C": 100,
    "D": 500,
    "M": 1000,
}

# Canonical form: thousands (M{0,3}), hundreds, tens, units.
_CANONICAL = re.compile(
    r"M{0,3}"
    r"(?:CM|CD|D?C{0,3})"
    r"(?:XC|XL|L?X{0,3})"
    r"(?:IX|IV|V?I{0,3})",
    re.ASCII,
)


def to_roman(value: int) -> str:
    """Convert an integer in 1..3999 to a canonical uppercase Roman numeral.

    Raises ValueError for bools, non-integers, and out-of-range values.
    """
    if isinstance(value, bool) or not isinstance(value, int):
        raise ValueError("value must be an int (not bool) in range 1..3999")
    if value < _MIN or value > _MAX:
        raise ValueError("value must be in range 1..3999, got %r" % (value,))

    remaining = value
    parts = []
    for amount, symbol in _VALUES:
        if remaining == 0:
            break
        count, remaining = divmod(remaining, amount)
        if count:
            parts.append(symbol * count)
    return "".join(parts)


def from_roman(text: str) -> int:
    """Convert a canonical uppercase Roman numeral (1..3999) to an int.

    Only nonempty, strictly canonical uppercase numerals are accepted.
    Whitespace, lowercase letters, unknown symbols, invalid repetitions,
    noncanonical subtractive forms and non-str inputs raise ValueError.
    """
    if not isinstance(text, str):
        raise ValueError("text must be a str")
    if not text:
        raise ValueError("text must be a nonempty Roman numeral")
    if _CANONICAL.fullmatch(text) is None:
        raise ValueError("invalid Roman numeral: %r" % (text,))

    total = 0
    n = len(text)
    i = 0
    while i < n:
        current = _SYMBOLS[text[i]]
        if i + 1 < n and _SYMBOLS[text[i + 1]] > current:
            total += _SYMBOLS[text[i + 1]] - current
            i += 2
        else:
            total += current
            i += 1

    # Defensive round-trip check guarantees strict canonical acceptance.
    if total < _MIN or total > _MAX or to_roman(total) != text:
        raise ValueError("invalid Roman numeral: %r" % (text,))
    return total
```
